cmake_minimum_required(VERSION 3.20)
project(llm_inference_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(llm_engine src/engine.cpp)
target_include_directories(llm_engine PUBLIC include)
if(MSVC)
  target_compile_options(llm_engine PRIVATE /O2)
else()
  target_compile_options(llm_engine PRIVATE -O3 -pthread)
endif()
add_executable(rfg_llm_demo src/main.cpp)
target_link_libraries(rfg_llm_demo PRIVATE llm_engine)
